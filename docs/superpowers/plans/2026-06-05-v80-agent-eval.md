```python
"""Trace 日志:agent 决策过程累积。

设计:
- Step 是不可变 dataclass,记录一步决策(type/round/model/tool/duration/error)
- TraceStore 累积 Step,max_steps 防内存爆炸
- to_dict() 序列化给 SSE done 事件透传
"""
from __future__ import annotations

from collections import deque
from dataclasses import dataclass, field
from typing import Any


@dataclass
class Step:
    """单步 agent 决策。

    type 取值:
    - "llm_call"     一次 LLM 调用
    - "tool_call"    一次 tool dispatch
    - "tool_result"  tool 返值预览(目前合并到 tool_call,留扩展)
    - "fallback"     LLM 失败降级到下一项
    - "retry"        反射重试
    - "done"         agent 结束
    """
    round: int
    type: str
    model_used: str | None = None
    tool_name: str | None = None
    tool_args: dict | None = None
    tool_result_preview: str | None = None
    duration_ms: int = 0
    error: str | None = None
    metadata: dict = field(default_factory=dict)

    def to_dict(self) -> dict[str, Any]:
        return {
            "round": self.round,
            "type": self.type,
            "model_used": self.model_used,
            "tool_name": self.tool_name,
            "tool_args": self.tool_args,
            "tool_result_preview": self.tool_result_preview,
            "duration_ms": self.duration_ms,
            "error": self.error,
            "metadata": self.metadata,
        }


class TraceStore:
    """累积 Step 的 bounded queue。"""

    def __init__(self, max_steps: int = 200):
        self._max = max_steps
        self._steps: deque[Step] = deque(maxlen=max_steps)

    def append(self, step: Step) -> None:
        self._steps.append(step)

    def get_all(self) -> list[dict[str, Any]]:
        return [s.to_dict() for s in self._steps]

    def clear(self) -> None:
        self._steps.clear()
```

```python
"""多模型 fallback 链 — LLM 失败自动降级到下一项。

设计:
- DEFAULT_FALLBACK_CHAIN: 5 个模型(qwen-max → qwen-long → qwen-coder-turbo → deepseek-v3 → kimi-k2.5)
- 错误分类:fatal(401/403/404,立即抛) vs retryable(timeout/429/5xx,降级)
- run_with_fallback(): 循环 chain,成功立刻返;链全失败抛 FallbackError
"""
from __future__ import annotations

import logging
from dataclasses import dataclass
from typing import Any

from backend.ai import llm_client

logger = logging.getLogger(__name__)


@dataclass(frozen=True)
class ModelEntry:
    """fallback 链中的一项。"""
    backend: str      # "openai" / "anthropic" / "ollama"
    model: str        # LLM 模型名
    priority: int     # 0=首选,1=次选...
    label: str = ""   # 人读名(可选,默认 backend:model)

    def __str__(self) -> str:
        return self.label or f"{self.backend}:{self.model}"


# 来自 docs/AI_MODEL_REGISTRY.md 实测 25 个全功能模型,挑 5 个有代表性的
DEFAULT_FALLBACK_CHAIN: list[ModelEntry] = [
    ModelEntry("openai", "qwen-max",         0, "qwen-max (默认)"),
    ModelEntry("openai", "qwen-long",        1, "qwen-long (长文本)"),
    ModelEntry("openai", "qwen-coder-turbo", 2, "qwen-coder-turbo (代码)"),
    ModelEntry("openai", "deepseek-v3",      3, "deepseek-v3 (第三方)"),
    ModelEntry("openai", "kimi-k2.5",        4, "kimi-k2.5 (兜底)"),
]


# 错误分类
class LLMError(Exception):
    """LLM 通用错误。"""


class FatalLLMError(LLMError):
    """不可重试错误(401/403/404)— 立即抛,不降级。"""


class FallbackError(LLMError):
    """fallback 链全失败,聚合每项错误。"""


# 可重试的错误信号(子串匹配,小写)
_RETRYABLE_SIGNALS = ("timeout", "rate limit", "free tier", "exhausted",
                      "429", "500", "502", "503", "504", "connection")
_FATAL_SIGNALS = ("401", "unauthorized", "403", "forbidden", "404", "not found",
                  "invalid api key", "permission denied")


def classify_error(err: Exception) -> str:
    """根据错误信息分类:fatal / retryable。

    简单字符串匹配,生产可换成 litellm 异常类型判断。
    """
    msg = str(err).lower()
    for sig in _FATAL_SIGNALS:
        if sig in msg:
            return "fatal"
    return "retryable"  # 默认可重试,降级更安全


def _build_cfg_for_entry(entry: ModelEntry):
    """根据 ModelEntry 构造一个临时 AIConfig(用当前 cfg 的其他参数 + 切换 backend/model)。"""
    from backend.ai.config import AIConfig, load_config
    base = load_config()
    return AIConfig(
        default_backend=base.default_backend,
        active_backend=entry.backend,
        model=entry.model,
        api_key=base.api_key,
        base_url=base.base_url,
        temperature=base.temperature,
        max_tokens=base.max_tokens,
        timeout_sec=base.timeout_sec,
        max_tool_rounds=base.max_tool_rounds,
    )


def run_with_fallback(
    messages: list[dict],
    tools: list[dict] | None = None,
    chain: list[ModelEntry] | None = None,
) -> tuple[Any, ModelEntry]:
    """循环 fallback 链,成功立刻返 (LLMResponse, ModelEntry);失败抛 FallbackError/FatalLLMError。

    可重试错误(timeout/429/...)→ 降级下一项
    致命错误(401/403/404)→ 立即抛 FatalLLMError
    链全失败 → 抛 FallbackError(聚合所有错误)
    """
    chain = chain or DEFAULT_FALLBACK_CHAIN
    errors: list[str] = []

    for entry in chain:
        try:
            cfg = _build_cfg_for_entry(entry)
            resp = llm_client.chat(cfg, messages=messages, tools=tools)
            if entry != chain[0]:
                logger.info("fallback succeeded: %s", entry)
            return resp, entry
        except llm_client.LLMError as e:
            category = classify_error(e)
            err_str = f"{entry.model}: {type(e).__name__}: {e}"
            errors.append(err_str)
            logger.warning("fallback chain entry %s failed [%s]: %s", entry, category, e)
            if category == "fatal":
                raise FatalLLMError(f"致命错误,不降级: {err_str}") from e
            # retryable: 继续下一项
            continue
        except Exception as e:
            err_str = f"{entry.model}: {type(e).__name__}: {e}"
            errors.append(err_str)
            logger.warning("fallback chain entry %s unexpected error: %s", entry, e)
            continue

    raise FallbackError(f"fallback 链 {len(chain)} 项全失败:\n  " + "\n  ".join(errors))
```

```python
    assert "error" in results[1]
```

```python
"""并行 tool 执行 — ThreadPoolExecutor 包装。

设计:
- 默认 max_workers=5(9 个 tool 不会一次全调)
- 顺序保持(用 enumerate 索引)
- 异常隔离(单个失败不影响其他,异常包装成 {"error": "..."})
- tool_timeout_sec 单 tool 超时(从 cfg 读,默认 30s)
"""
from __future__ import annotations

import logging
from concurrent.futures import ThreadPoolExecutor, TimeoutError as FuturesTimeoutError
from typing import Any

from backend.ai.llm_client import ToolCall
from backend.ai.tool_executor import execute_tool

logger = logging.getLogger(__name__)


def execute_tools_parallel(
    tool_calls: list[ToolCall],
    max_workers: int = 5,
    tool_timeout_sec: int = 30,
) -> list[dict[str, Any]]:
    """并行执行 tool_call 列表,返与入参同序的 result dict 列表。"""
    if not tool_calls:
        return []

    results: list[dict[str, Any] | None] = [None] * len(tool_calls)
    executor = ThreadPoolExecutor(max_workers=min(max_workers, len(tool_calls)))

    def _run(idx: int, tc: ToolCall) -> None:
        try:
            results[idx] = execute_tool(tc.name, tc.arguments)
        except Exception as e:
            logger.warning("tool %s failed in parallel: %s", tc.name, e)
            results[idx] = {
                "error": f"{tc.name}: {type(e).__name__}: {e}",
                "navigate_to": None,
            }

    futures = [executor.submit(_run, i, tc) for i, tc in enumerate(tool_calls)]
    # 等全部完成(带超时)
    for i, future in enumerate(futures):
        try:
            future.result(timeout=tool_timeout_sec)
        except FuturesTimeoutError:
            logger.warning("tool %s timeout after %ss", tool_calls[i].name, tool_timeout_sec)
            results[i] = {
                "error": f"{tool_calls[i].name}: timeout after {tool_timeout_sec}s",
                "navigate_to": None,
            }
        except Exception as e:
            # _run 已捕获,但 double-check
            if results[i] is None:
                results[i] = {"error": f"{tool_calls[i].name}: {e}", "navigate_to": None}

    executor.shutdown(wait=True)
    return results  # type: ignore[return-value]
```

```python
"""反射重试 — tool 失败时构造 hint 引导 LLM 改参数重试。

设计:
- RETRYABLE_TOOL_ERRORS 白名单:可重试的 tool 错误类型
- should_retry(result): 检查 tool 返的 error 字段是否在白名单
- build_retry_hint(tool_name, args, error): 构造 user 消息 hint

重试策略:
- 单 tool 失败 → chat_service 在同一 round 内追加 user hint
- LLM 下一轮会看到 hint + 之前的 tool result,自动改参数再调
- 最多 1 次反射重试(避免无限循环)— 在 chat_service 层面控制
"""
from __future__ import annotations

import json
from typing import Any


# 可重试的 tool 错误白名单(这些错误通常是参数问题,改参数可解)
RETRYABLE_TOOL_ERRORS: set[str] = {
    "city_not_found",          # 城市名拼写错(中文/拼音)
    "data_unavailable",        # 数据缺失,换 scope/indicator
    "missing_required_fields", # 缺必填参数
    "timeout",                 # tool 执行超时
}


# 错误类型 → 修复建议(中英混合,因为 LLM 看得懂)
_HINTS: dict[str, str] = {
    "city_not_found": "请检查城市名是否正确(中文/拼音),可调用 list_cities 获取支持的列表。",
    "data_unavailable": "请尝试换 scope(province/city/county)或换 indicator,或减少 years。",
    "missing_required_fields": "请检查并补充所有必填参数(参考 tool schema)。",
    "timeout": "请缩小查询范围(如减少 years)后重试。",
}


def should_retry(tool_result: dict[str, Any]) -> bool:
    """tool 返的 error 字段在 RETRYABLE_TOOL_ERRORS 白名单 → True。"""
    err = tool_result.get("error")
    if not err:
        return False
    # 匹配白名单子串(支持 "city_not_found: 深圳" 这种带 context 的)
    return any(e in str(err) for e in RETRYABLE_TOOL_ERRORS)


def build_retry_hint(tool_name: str, original_args: dict[str, Any], error: str) -> str:
    """构造 user 消息 hint 引导 LLM 改参数重试。"""
    # 找匹配的 hint 模板
    template = None
    for key, text in _HINTS.items():
        if key in error:
            template = text
            break
    if not template:
        template = "请检查 tool 参数是否正确后重试。"

    args_str = json.dumps(original_args, ensure_ascii=False, default=str)[:300]
    return (
        f"[系统提示] tool `{tool_name}` 调用失败(错误: {error})。\n"
        f"原参数: {args_str}\n"
        f"建议: {template}"
    )
```

```python
def chat_stream_with_trace(
    cfg: AIConfig,
    session_id: str,
    user_message: str,
) -> Any:
    """流式 chat — v8.0:多模型 fallback + 并行 tool + trace + 反射重试。

    事件类型:
    - {"type": "thinking"}                          启动
    - {"type": "tool_call", "name", "args", ...}    一次 tool dispatch
    - {"type": "token", "content": "..."}           文本片段
    - {"type": "done", "finish_reason", "trace":[]} 结束(含 trace)
    - {"type": "error", "message", "trace":[]}      失败
    """
    import time
    yield {"type": "thinking"}
    messages = _build_messages(session_id, user_message)
    trace = TraceStore()
    tool_rounds = 0
    total_prompt_tokens = 0
    total_completion_tokens = 0
    max_rounds = getattr(cfg, "max_tool_rounds", 5) if cfg is not None else 5
    tool_timeout = getattr(cfg, "tool_timeout_sec", 30)
    max_workers = getattr(cfg, "parallel_max_workers", 5)
    chain = getattr(cfg, "fallback_chain", None) or None

    try:
        for _ in range(max_rounds + 1):
            # 1. LLM (带 fallback)
            t0 = time.perf_counter()
            try:
                resp, used_model = run_with_fallback(messages, TOOLS, chain)
            except FatalLLMError as e:
                trace.append(Step(type="llm_call", error=str(e),
                                  duration_ms=int((time.perf_counter()-t0)*1000)))
                yield {"type": "error", "message": str(e), "trace": trace.get_all()}
                return
            except FallbackError as e:
                trace.append(Step(type="llm_call", error=str(e),
                                  duration_ms=int((time.perf_counter()-t0)*1000)))
                yield {"type": "error", "message": f"所有模型都不可用: {e}", "trace": trace.get_all()}
                return

            total_prompt_tokens += resp.prompt_tokens
            total_completion_tokens += resp.completion_tokens
            trace.append(Step(
                round=tool_rounds + 1,
                type="llm_call",
                model_used=used_model.model,
                duration_ms=int((time.perf_counter()-t0)*1000),
                metadata={"prompt_tokens": resp.prompt_tokens, "completion_tokens": resp.completion_tokens},
            ))

            if not resp.tool_calls:
                # 最终答案
                content = resp.content or ""
                yield {"type": "token", "content": content}
                conversation.append_message(session_id, {"role": "user", "content": user_message})
                conversation.append_message(session_id, {"role": "assistant", "content": content})
                trace.append(Step(round=tool_rounds+1, type="done",
                                  metadata={"reason": "stop", "total_tokens": total_prompt_tokens+total_completion_tokens}))
                yield {
                    "type": "done",
                    "finish_reason": "stop",
                    "tool_rounds": tool_rounds,
                    "tool_calls": [],
                    "trace": trace.get_all(),
                }
                return

            # 2. LLM 决策加进 messages
            messages.append({
                "role": "assistant",
                "content": resp.content or "",
                "tool_calls": [
                    {
                        "id": tc.id, "type": "function",
                        "function": {"name": tc.name, "arguments": json.dumps(tc.arguments, ensure_ascii=False)},
                    }
                    for tc in resp.tool_calls
                ],
            })

            # 3. 并行执行 tool
            t1 = time.perf_counter()
            results = execute_tools_parallel(resp.tool_calls, max_workers=max_workers, tool_timeout_sec=tool_timeout)
            parallel_ms = int((time.perf_counter()-t1)*1000)
            for tc, result in zip(resp.tool_calls, results):
                preview = json.dumps(result.get("data") or result, ensure_ascii=False, default=str)[:200]
                tool_event = {
                    "type": "tool_call",
                    "name": tc.name,
                    "args": tc.arguments,
                    "navigate_to": result.get("navigate_to"),
                    "data_preview": preview if "data" in result else None,
                    "error": result.get("error"),
                }
                yield tool_event
                trace.append(Step(
                    round=tool_rounds+1, type="tool_call",
                    tool_name=tc.name, tool_args=tc.arguments,
                    tool_result_preview=preview,
                    duration_ms=parallel_ms // max(1, len(resp.tool_calls)),
                    error=result.get("error"),
                ))

            # 4. 反射重试 hint(单 tool 失败)
            for tc, result in zip(resp.tool_calls, results):
                if should_retry(result):
                    hint = build_retry_hint(tc.name, tc.arguments, result["error"])
                    messages.append({"role": "user", "content": hint})
                    trace.append(Step(type="retry", tool_name=tc.name, error=result["error"]))

            # 5. tool results 进 messages
            for tc, result in zip(resp.tool_calls, results):
                messages.append({
                    "role": "tool", "tool_call_id": tc.id, "name": tc.name,
                    "content": json.dumps(result, ensure_ascii=False, default=str),
                })

            tool_rounds += 1
            if tool_rounds >= max_rounds:
                messages.append({"role": "user", "content": "请基于已有 tool 结果直接给最终答案,不要再调 tool。"})
                try:
                    resp2, used_model2 = run_with_fallback(messages, TOOLS, chain)
                    content = resp2.content or "(问题过于复杂,无法在限定步数内回答)"
                except (FatalLLMError, FallbackError) as e:
                    content = f"(强制总结时 LLM 失败: {e})"
                yield {"type": "token", "content": content}
                conversation.append_message(session_id, {"role": "user", "content": user_message})
                conversation.append_message(session_id, {"role": "assistant", "content": content})
                trace.append(Step(round=tool_rounds+1, type="done", metadata={"reason": "tool_rounds_exhausted", "forced_stop": True, "total_tokens": total_prompt_tokens+total_completion_tokens}))
                yield {
                    "type": "done", "finish_reason": "tool_rounds_exhausted",
                    "tool_rounds": tool_rounds, "forced_stop": True, "tool_calls": [],
                    "trace": trace.get_all(),
                }
                return

        yield {
            "type": "done", "finish_reason": "max_rounds",
            "tool_rounds": tool_rounds, "tool_calls": [],
            "trace": trace.get_all(),
        }
    except LLMError as e:
        logger.warning("LLM error in chat_stream: %s", e)
        yield {"type": "error", "message": str(e), "trace": trace.get_all()}
```

`tests/eval/scorer.py`:

```python
"""Eval 评分 — 6 维度,每 case 0-100 分。

维度(权重):
- tool 选择(30)   集合完全匹配 = 30,部分匹配按比例
- args 匹配(20)   expected_args_match 全 key 命中 = 20,按命中比例
- 关键词(20)      expected_keywords 命中 >= 50% = 20,按比例
- 延迟(15)        < 10s = 15,< 30s = 10,< 60s = 5,>= 60s = 0
- fallback(10)     0 次 = 10,1 次 = 5,>= 2 次 = 0
- token 效率(5)   < 2K = 5,< 5K = 3,>= 5K = 1
"""
from __future__ import annotations

import time
from dataclasses import dataclass
from typing import Any


@dataclass
class CaseResult:
    case_id: str
    query: str
    actual_tools: list[str]
    actual_args: list[dict]
    actual_answer: str
    actual_fallback_count: int
    actual_total_tokens: int
    actual_duration_sec: float
    score: float
    breakdown: dict[str, float]
    pass_: bool


def score_case(
    case: dict[str, Any],
    actual_tools: list[str],
    actual_args: list[dict],
    actual_answer: str,
    actual_fallback_count: int,
    actual_total_tokens: int,
    actual_duration_sec: float,
) -> CaseResult:
    """单 case 评分。"""
    breakdown: dict[str, float] = {}

    # 1. tool 选择(30)
    expected_tools = set(case["expected_tools"])
    actual_tools_set = set(actual_tools)
    if expected_tools:
        tool_match_ratio = len(expected_tools & actual_tools_set) / len(expected_tools)
    else:
        tool_match_ratio = 1.0
    breakdown["tool"] = round(tool_match_ratio * 30, 2)

    # 2. args 匹配(20)
    expected_args = case.get("expected_args_match", {})
    if expected_args:
        # 任一 actual_args 包含 expected_args 所有 key+value → 满分
        max_match = 0
        for a in actual_args:
            matches = sum(1 for k, v in expected_args.items() if a.get(k) == v)
            ratio = matches / len(expected_args)
            max_match = max(max_match, ratio)
        breakdown["args"] = round(max_match * 20, 2)
    else:
        breakdown["args"] = 20.0

    # 3. 关键词(20)
    expected_kw = case.get("expected_keywords", [])
    if expected_kw:
        kw_hits = sum(1 for kw in expected_kw if kw in actual_answer)
        kw_ratio = kw_hits / len(expected_kw)
        breakdown["keywords"] = round(min(1.0, kw_ratio * 2) * 20, 2)  # 50% 命中即满分
    else:
        breakdown["keywords"] = 20.0

    # 4. 延迟(15)
    if actual_duration_sec < 10:
        breakdown["latency"] = 15.0
    elif actual_duration_sec < 30:
        breakdown["latency"] = 10.0
    elif actual_duration_sec < 60:
        breakdown["latency"] = 5.0
    else:
        breakdown["latency"] = 0.0

    # 5. fallback(10)
    if actual_fallback_count == 0:
        breakdown["fallback"] = 10.0
    elif actual_fallback_count == 1:
        breakdown["fallback"] = 5.0
    else:
        breakdown["fallback"] = 0.0

    # 6. token 效率(5)
    if actual_total_tokens < 2000:
        breakdown["tokens"] = 5.0
    elif actual_total_tokens < 5000:
        breakdown["tokens"] = 3.0
    else:
        breakdown["tokens"] = 1.0

    total = sum(breakdown.values())
    # pass 条件:总分 >= 70 AND tool 分 >= 20
    pass_ = total >= 70 and breakdown["tool"] >= 20

    return CaseResult(
        case_id=case["id"],
        query=case["query"],
        actual_tools=actual_tools,
        actual_args=actual_args,
        actual_answer=actual_answer,
        actual_fallback_count=actual_fallback_count,
        actual_total_tokens=actual_total_tokens,
        actual_duration_sec=actual_duration_sec,
        score=round(total, 2),
        breakdown=breakdown,
        pass_=pass_,
    )
```

```python
"""Eval 报表生成 — JSON + Markdown 两种格式。

输出:tests/eval/reports/<date>-<model>.{json,md}
"""
from __future__ import annotations

import json
from datetime import datetime
from pathlib import Path
from typing import Any

from tests.eval.scorer import CaseResult

REPORTS_DIR = Path(__file__).parent / "reports"


def _summary(results: list[CaseResult]) -> dict[str, Any]:
    if not results:
        return {"avg_score": 0, "pass_rate": 0, "total_cases": 0}
    avg = sum(r.score for r in results) / len(results)
    pass_rate = sum(1 for r in results if r.pass_) / len(results)
    return {
        "avg_score": round(avg, 2),
        "pass_rate": round(pass_rate, 3),
        "total_cases": len(results),
    }


def save_report(
    model: str,
    results: list[CaseResult],
    date_str: str | None = None,
) -> Path:
    """保存 JSON 报表。"""
    REPORTS_DIR.mkdir(parents=True, exist_ok=True)
    date_str = date_str or datetime.now().strftime("%Y-%m-%d")
    path = REPORTS_DIR / f"{date_str}-{model}.json"
    payload = {
        "model": model,
        "date": date_str,
        "summary": _summary(results),
        "cases": [
            {
                "id": r.case_id, "query": r.query,
                "actual_tools": r.actual_tools, "actual_args": r.actual_args,
                "actual_answer": r.actual_answer[:300],
                "actual_fallback_count": r.actual_fallback_count,
                "actual_total_tokens": r.actual_total_tokens,
                "actual_duration_sec": round(r.actual_duration_sec, 2),
                "score": r.score, "breakdown": r.breakdown, "pass": r.pass_,
            }
            for r in results
        ],
    }
    path.write_text(json.dumps(payload, ensure_ascii=False, indent=2))
    return path


def save_comparison(
    models: list[str],
    all_results: dict[str, list[CaseResult]],
    date_str: str | None = None,
) -> Path:
    """保存多模型对比报表(每个模型一份 JSON,加一份总览 MD)。"""
    REPORTS_DIR.mkdir(parents=True, exist_ok=True)
    date_str = date_str or datetime.now().strftime("%Y-%m-%d")
    # 每个模型独立 JSON
    for m, results in all_results.items():
        save_report(m, results, date_str)
    # 总览 MD
    md_path = REPORTS_DIR / f"{date_str}-compare.md"
    lines = [f"# Eval 对比报告 — {date_str}", ""]
    lines.append("| 模型 | 平均分 | 通过率 | Case 数 |")
    lines.append("|------|--------|--------|---------|")
    for m, results in all_results.items():
        s = _summary(results)
        lines.append(f"| {m} | {s['avg_score']} | {s['pass_rate']*100:.0f}% | {s['total_cases']} |")
    lines.append("")
    md_path.write_text("\n".join(lines))
    return md_path
```

```python
"""Eval CLI 入口。

用法:
    # 单模型
    python -m tests.eval.run_eval --model qwen-max

    # 多模型对比
    python -m tests.eval.run_eval --compare qwen-max deepseek-v3 kimi-k2.5

    # 跑指定 case 子集
    python -m tests.eval.run_eval --model qwen-max --ids shenzhen-gdp-forecast-3y
"""
from __future__ import annotations

import argparse
import json
import os
import sys
import time
from pathlib import Path

from backend.ai.config import AIConfig
from backend.ai.fallback import ModelEntry, DEFAULT_FALLBACK_CHAIN
from backend.ai import chat_service
from backend.ai.tool_registry import TOOLS
from tests.eval.scorer import score_case, CaseResult
from tests.eval.reporter import save_report, save_comparison


def _load_cases(path: Path, ids: list[str] | None = None) -> list[dict]:
    cases = [json.loads(line) for line in path.read_text().splitlines() if line.strip()]
    if ids:
        cases = [c for c in cases if c["id"] in ids]
    return cases


def _cfg_for_model(model: str) -> AIConfig:
    """构造一个只含指定模型的 AIConfig + fallback chain 仅含自己(避免链干扰评分)。"""
    base = AIConfig(
        default_backend="openai",
        active_backend="openai",
        model=model,
        api_key=os.environ.get("LLM_OPENAI_API_KEY", "test"),
        base_url=os.environ.get("LLM_OPENAI_BASE_URL", "https://dashscope.aliyuncs.com/compatible-mode/v1"),
        temperature=0.3,
        max_tokens=2000,
        timeout_sec=60,
        max_tool_rounds=5,
        fallback_chain=[ModelEntry("openai", model, 0, model)],
    )
    return base


def _run_case(case: dict, cfg: AIConfig) -> CaseResult:
    """跑单 case,收集实际 tool/answer/fallback/tokens/duration,评分。"""
    actual_tools: list[str] = []
    actual_args: list[dict] = []
    fallback_count = 0
    total_tokens = 0
    answer_parts: list[str] = []
    t0 = time.perf_counter()
    try:
        for ev in chat_service.chat_stream_with_trace(cfg, f"eval-{case['id']}", case["query"]):
            if ev["type"] == "tool_call":
                actual_tools.append(ev["name"])
                actual_args.append(ev.get("args") or {})
            elif ev["type"] == "token":
                answer_parts.append(ev.get("content", ""))
            elif ev["type"] == "done":
                # 从 trace 统计 fallback 和 tokens
                for step in ev.get("trace", []):
                    if step.get("type") == "llm_call" and step.get("model_used") != cfg.model:
                        fallback_count += 1
                    total_tokens += step.get("metadata", {}).get("prompt_tokens", 0)
                    total_tokens += step.get("metadata", {}).get("completion_tokens", 0)
    except Exception as e:
        answer_parts.append(f"(eval error: {e})")
    duration = time.perf_counter() - t0
    return score_case(
        case=case,
        actual_tools=actual_tools,
        actual_args=actual_args,
        actual_answer="".join(answer_parts),
        actual_fallback_count=fallback_count,
        actual_total_tokens=total_tokens,
        actual_duration_sec=duration,
    )


def main() -> int:
    parser = argparse.ArgumentParser(description="Run eval against golden cases")
    parser.add_argument("--model", help="single model to evaluate")
    parser.add_argument("--compare", nargs="+", help="multiple models to compare")
    parser.add_argument("--ids", nargs="+", help="filter to specific case ids")
    args = parser.parse_args()

    cases_path = Path(__file__).parent / "golden_cases.jsonl"
    cases = _load_cases(cases_path, args.ids)
    print(f"Loaded {len(cases)} cases from {cases_path}")

    models = [args.model] if args.model else args.compare
    if not models:
        models = [e.model for e in DEFAULT_FALLBACK_CHAIN]
    print(f"Models: {models}")

    all_results: dict[str, list[CaseResult]] = {}
    for model in models:
        print(f"\n=== {model} ===")
        cfg = _cfg_for_model(model)
        results = []
        for case in cases:
            r = _run_case(case, cfg)
            marker = "PASS" if r.pass_ else "FAIL"
            print(f"  {marker} {case['id']:35s} score={r.score:5.1f} tools={r.actual_tools}")
            results.append(r)
        all_results[model] = results
        report_path = save_report(model, results)
        print(f"  Report: {report_path}")

    if len(all_results) > 1:
        cmp_path = save_comparison(list(all_results.keys()), all_results)
        print(f"\nComparison: {cmp_path}")

    return 0


if __name__ == "__main__":
    sys.exit(main())
```
