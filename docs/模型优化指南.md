### 2.1 评估指标

#### 回归任务

```python
from sklearn.metrics import (
    mean_absolute_error,           # MAE
    mean_squared_error,            # MSE
    r2_score,                     # R²
    mean_absolute_percentage_error # MAPE
)

def evaluate_model(y_true, y_pred):
    """完整评估"""
    metrics = {
        'MAE': mean_absolute_error(y_true, y_pred),
        'MSE': mean_squared_error(y_true, y_pred),
        'RMSE': np.sqrt(mean_squared_error(y_true, y_pred)),
        'R2': r2_score(y_true, y_pred),
        'MAPE': mean_absolute_percentage_error(y_true, y_pred) * 100
    }
    return metrics
```

### 2.2 模型选择标准

```python
def select_best_model(result, criteria='r2'):
    """选择最佳模型"""
    if criteria == 'r2':
        # 选择R²最高的模型
        best = max(result.ml_result.all_results, 
                   key=lambda x: x.metrics.r2)
    elif criteria == 'mae':
        # 选择MAE最低的模型
        best = min(result.ml_result.all_results, 
                   key=lambda x: x.metrics.mae)
    elif criteria == 'balance':
        # 平衡R²和MAE
        def score(x):
            return x.metrics.r2 / (1 + x.metrics.mae / 100)
        best = max(result.ml_result.all_results, key=score)
    
    return best
```

## 3. 模型监控与迭代

### 3.1 性能监控

```python
import json
from datetime import datetime

class ModelMonitor:
    """模型性能监控"""
    
    def __init__(self, model_name):
        self.model_name = model_name
        self.history = []
    
    def log_performance(self, metrics, timestamp=None):
        """记录性能指标"""
        if timestamp is None:
            timestamp = datetime.now().isoformat()
        
        self.history.append({
            'timestamp': timestamp,
            'metrics': metrics
        })
    
    def check_drift(self, window=10):
        """检测性能漂移"""
        if len(self.history) < window:
            return False
        
        recent = self.history[-window:]
        old = self.history[-2*window:-window] if len(self.history) >= 2*window else recent
        
        old_r2 = np.mean([h['metrics']['r2'] for h in old])
        recent_r2 = np.mean([h['metrics']['r2'] for h in recent])
        
        # 如果R²下降超过10%，认为有性能漂移
        drift_ratio = (old_r2 - recent_r2) / old_r2
        return drift_ratio > 0.10
    
    def get_summary(self):
        """获取性能摘要"""
        if not self.history:
            return "No data"
        
        r2_values = [h['metrics']['r2'] for h in self.history]
        
        return {
            'total_runs': len(self.history),
            'avg_r2': np.mean(r2_values),
            'std_r2': np.std(r2_values),
            'best_r2': np.max(r2_values),
            'latest_r2': r2_values[-1]
        }
```

### 3.2 模型更新流程

```python
class ModelUpdater:
    """模型更新管理器"""
    
    def __init__(self, analyzer, model_path='./models'):
        self.analyzer = analyzer
        self.model_path = Path(model_path)
        self.model_path.mkdir(exist_ok=True)
    
    def update_model(self, new_data, retrain=True):
        """更新模型"""
        if retrain:
            print("正在重新训练模型...")
            
            result = self.analyzer.analyze(
                run_ml=True,
                target_column=self.analyzer.config.ml_config.target_metric,
                feature_columns=self.analyzer.config.ml_config.feature_columns,
                hyperparameter_tuning=True
            )
            
            # 保存新模型
            best_model = result.ml_result.best_model.model
            model_file = self.model_path / f"{result.ml_result.best_model.model_name}_updated.pkl"
            best_model.save(str(model_file))
            
            print(f"✓ 模型已更新并保存到 {model_file}")
            
            return result
        else:
            print("使用现有模型...")
            return None
    
    def rollback(self, backup_file):
        """回滚到旧模型"""
        import joblib
        model_data = joblib.load(backup_file)
        print(f"✓ 已回滚到 {backup_file}")
        return model_data
```

```python
def analyze_feature_importance(result):
    """分析特征重要性"""
    
    print("\n特征重要性分析：")
    print("=" * 60)
    
    importance = result.ml_result.best_model.feature_importance
    
    if not importance:
        print("特征重要性不可用（模型不支持）")
        return
    
    # 排序
    sorted_features = sorted(importance.items(), 
                            key=lambda x: x[1], 
                            reverse=True)
    
    print(f"\n{'特征':<30} {'重要性':<15} {'累计贡献':<10}")
    print("-" * 60)
    
    total_importance = sum(importance.values())
    cumulative = 0
    
    for feat, imp in sorted_features:
        cumulative += imp
        contribution = imp / total_importance * 100
        
        print(f"{feat:<30} {imp:<15.4f} {contribution:<10.2f}%")
    
    # 识别关键特征（贡献>5%）
    print("\n关键特征（贡献>5%）：")
    for feat, imp in sorted_features:
        if imp / total_importance * 100 > 5:
            print(f"  ✓ {feat}: {imp:.4f}")
```

```python
def suggest_feature_optimization(result):
    """特征优化建议"""
    
    importance = result.ml_result.best_model.feature_importance
    
    suggestions = {
        'high_importance': [],
        'low_importance': [],
        'potential_new_features': []
    }
    
    for feat, imp in importance.items():
        if imp > 0.1:
            suggestions['high_importance'].append(feat)
            suggestions['potential_new_features'].append(f"{feat}_lag1")  # 滞后特征
            suggestions['potential_new_features'].append(f"{feat}_growth")  # 增长率
        else:
            suggestions['low_importance'].append(feat)
    
    print("\n特征优化建议：")
    print("1. 高重要性特征（需重点关注）：")
    for feat in suggestions['high_importance']:
        print(f"   - {feat}")
    
    print("\n2. 低重要性特征（可考虑移除）：")
    for feat in suggestions['low_importance']:
        print(f"   - {feat}")
    
    print("\n3. 建议添加的新特征：")
    for feat in suggestions['potential_new_features'][:5]:
        print(f"   - {feat}")
    
    return suggestions
```
