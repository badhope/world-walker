# v9.0 Python SDK + CLI Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a Python SDK (`rca/`) and Typer CLI (`cli/`) so users can call all 9 agent tools and the LLM agent from Python imports and command line, in addition to the existing Web chat widget.

**Architecture:** SDK wraps existing `backend.ai.tool_executor` (9 tool dicts) and `backend.ai.chat_service` (chat + chat_stream) as ergonomic Python functions and a `RCAClient` class. CLI uses Typer 0.12.3 (already in requirements-lock) with three output formatters (rich/json/plain). Errors propagate as `RCAError` hierarchy; CLI converts to exit codes 0/1/2/3.

**Tech Stack:** Typer 0.12.3, Click 8.1.7, Rich (Typer dep), pytest, pytest-mock. No new dependencies.

```python
# rca/errors.py
"""SDK 错误层次。

设计:
- RCAError 是所有 SDK 异常的基类,用户 except RCAError: 一次捕完
- ToolNotFoundError — 调了不存在的 tool 名
- ToolArgumentError — tool 必填字段缺失 / 值非法
- LLMUnavailableError — fallback 链全失败
- ConfigError — SDK 配置错(无 API key 等)
"""
from __future__ import annotations


class RCAError(Exception):
    """SDK 顶层错误。所有 SDK 抛出的异常都继承自本类。"""


class ToolNotFoundError(RCAError):
    """Tool 名错 — 调了不在 _DISPATCH 里的 tool。

    Example:
        RCAClient().invoke("get_foo", {})
        # → ToolNotFoundError: Tool not found: get_foo
    """

    def __init__(self, tool_name: str):
        self.tool_name = tool_name
        super().__init__(f"Tool not found: {tool_name}")


class ToolArgumentError(RCAError):
    """Tool 参数错 — 必填字段缺失 / tool 内部返回 {"error": "..."}。

    Example:
        RCAClient().invoke("get_forecast", {})  # 缺 name / scope / indicator
        # → ToolArgumentError: Missing required fields: ['scope', 'name', 'indicator']
    """


class LLMUnavailableError(RCAError):
    """所有 LLM 都不可用 — fallback 链全失败。

    Example:
        RCAClient().chat(...)  # 网络挂 / 全 401
        # → LLMUnavailableError: fallback 链 5 项全失败: qwen-max: ..., deepseek-v3: ...
    """


class ConfigError(RCAError):
    """SDK config 错(无 API key / 后端不存在等)。"""
```

# rca/_config.py
"""SDK 内部 config loader — 从 env 读 + 填默认 fallback chain + 默认 tool/parallel 配置。

设计:
- 不读 env(由调用方负责初始化),此函数读 env
- 把 AIConfig 的 3 个 v8.0 字段(fallback_chain, tool_timeout_sec, parallel_max_workers)填好
- 找不到任何 backend key → ConfigError
"""
from __future__ import annotations

from backend.ai.config import AIConfig
from backend.ai.fallback import DEFAULT_FALLBACK_CHAIN

from rca.errors import ConfigError


def load_config() -> AIConfig:
    """从 env 构造 AIConfig。

    Returns:
        AIConfig with fallback_chain = DEFAULT_FALLBACK_CHAIN,
        tool_timeout_sec = 30, parallel_max_workers = 5,
        其他字段从 env 读(load_config() from backend.ai.config)。

    Raises:
        ConfigError: 找不到任何 LLM backend 的 env var。
    """
    try:
        from backend.ai.config import load_config as _backend_load
        cfg = _backend_load()
    except Exception as e:
        raise ConfigError(str(e)) from e

    # 补 v8.0 字段(backend load_config() 已经填了默认值,但确保不空)
    if not cfg.fallback_chain:
        cfg = AIConfig(
            default_backend=cfg.default_backend,
            active_backend=cfg.active_backend,
            model=cfg.model,
            temperature=cfg.temperature,
            max_tokens=cfg.max_tokens,
            timeout_sec=cfg.timeout_sec,
            max_tool_rounds=cfg.max_tool_rounds,
            available_backends=cfg.available_backends,
            fallback_chain=tuple(DEFAULT_FALLBACK_CHAIN),
            tool_timeout_sec=cfg.tool_timeout_sec or 30,
            parallel_max_workers=cfg.parallel_max_workers or 5,
        )
    return cfg
```

# rca/tools.py
"""SDK 9 tool wrapper — 把 backend.ai.tool_executor 的 dict 接口包装成 **kwargs 函数。

设计:
- 每个 tool 1 个公开函数,**kwargs 跟 tool schema 一致
- 内部直接 import + 调 execute_tool(不重写逻辑)
- 错误处理:tool_executor 返 {"error": "..."} 时,抛 ToolArgumentError
"""
from __future__ import annotations

from typing import Any

from backend.ai import tool_executor

from rca.errors import ToolArgumentError, ToolNotFoundError


def _invoke(name: str, args: dict[str, Any]) -> dict[str, Any]:
    """统一 invoke — 处理 dispatch + 错误包装。

    Args:
        name: tool 名("get_forecast" 等)
        args: tool 参数字典

    Returns:
        tool_executor 返回的 dict(成功 {"data": ..., "navigate_to": ...})

    Raises:
        ToolNotFoundError: tool 名不在 _DISPATCH
        ToolArgumentError: tool 返 {"error": ...} 或缺必填字段
    """
    try:
        result = tool_executor.execute_tool(name, args)
    except KeyError as e:
        # execute_tool 内部不抛 KeyError,但 _DISPATCH 改了名要兼容
        raise ToolNotFoundError(name) from e

    if "error" in result:
        # tool_executor 把内部错包装成 {"error": "..."} — 转 ToolArgumentError 给 SDK
        raise ToolArgumentError(result["error"])

    return result


# --------------------------------------------------------------------------- #
# 9 tool wrapper — 名字对应 tool_registry 里的 snake_case
# --------------------------------------------------------------------------- #


def forecast(scope: str, name: str, indicator: str, years: int = 5) -> dict[str, Any]:
    """获取某城市/省份某指标的多年预测。"""
    return _invoke("get_forecast", {
        "scope": scope, "name": name, "indicator": indicator, "years": years,
    })
```

# rca/chat.py
"""SDK chat + chat_stream — 包装 backend.ai.chat_service,错误统一为 LLMUnavailableError。

设计:
- 模块级函数 chat() / chat_stream() — 不需要 RCAClient
- 内部 load_config() 拿 cfg(只在第一次调用时 load,后续复用)
- 任何 LLMError/FallbackError → LLMUnavailableError
- 返回值与 backend chat_service 100% 一致(不动结构)
"""
from __future__ import annotations

from typing import Any, Iterator

from backend.ai import chat_service as _chat_service
from backend.ai.fallback import FallbackError, FatalLLMError
from backend.ai.llm_client import LLMError

from rca._config import load_config
from rca.errors import LLMUnavailableError, RCAError

_cached_cfg = None


def _get_cfg():
    """懒加载 cfg(第一次调时 load,后续复用同一对象)。"""
    global _cached_cfg
    if _cached_cfg is None:
        _cached_cfg = load_config()
    return _cached_cfg


def chat(session_id: str, user_message: str) -> dict[str, Any]:
    """单轮 LLM agent — 调一次 LLM,允许多 tool round,返 final 答案。

    Args:
        session_id: 客户端会话 ID(用于 conversation store)
        user_message: 用户自然语言问题

    Returns:
        {"content": "...", "tool_rounds": N, "tool_calls": [...], "prompt_tokens": N, ...}

    Raises:
        LLMUnavailableError: 所有 LLM 都不可用
        RCAError: 其他 LLM 错(配置错 / 模型不存在 等)
    """
    try:
        return _chat_service.chat(_get_cfg(), session_id, user_message)
    except (FallbackError, FatalLLMError, LLMError) as e:
        raise LLMUnavailableError(str(e)) from e


def chat_stream(session_id: str, user_message: str) -> Iterator[dict[str, Any]]:
    """流式 LLM agent — yield 事件 dict,事件类型:
    - {"type": "thinking"}                          启动
    - {"type": "tool_call", "name", "args", ...}    一次 tool dispatch
    - {"type": "token", "content": "..."}           文本片段
    - {"type": "done", "finish_reason", "trace":[]} 结束(含 trace)
    - {"type": "error", "message", "trace":[]}      失败

    Yields:
        事件 dict
    """
    try:
        gen = _chat_service.chat_stream_with_trace(_get_cfg(), session_id, user_message)
        for ev in gen:
            yield ev
    except (FallbackError, FatalLLMError, LLMError) as e:
        # 不能 raise(generator 半路抛会断流);改 yield error event
        yield {"type": "error", "message": str(e), "trace": []}
```

"""RCAClient — SDK 顶层类,聚合 9 tools + chat + custom config。

设计:
- 默认 config: rca._config.load_config()(从 env 读)
- 传 config=... 可覆盖(适合多租户 / 多模型配置)
- 9 tool 方法纯委托给 rca.tools 模块(不重写)
- chat() / chat_stream() 委托给 rca.chat 模块

from __future__ import annotations

from typing import Any, Iterator, Optional

from backend.ai.config import AIConfig

from rca import chat as _chat
from rca import tools as _tools
from rca._config import load_config


class RCAClient:
    """SDK 顶层入口。

    用法:
        client = RCAClient()  # 读 env
        client = RCAClient(config=my_cfg)  # 自定义 cfg

        client.forecast(scope="city", name="shenzhen", indicator="gdp")
        client.chat("session-1", "深圳 GDP 2026 多少?")
        for ev in client.chat_stream("session-1", "..."):
            ...
    """

    def __init__(self, config: Optional[AIConfig] = None):
        """构造。

        Args:
            config: 自定义 AIConfig(覆盖 env)。None = load_config()。
        """
        self._cfg = config if config is not None else load_config()

    # 9 tool — 纯委托
    def forecast(self, scope: str, name: str, indicator: str, years: int = 5) -> dict[str, Any]:
        return _tools.forecast(scope=scope, name=name, indicator=indicator, years=years)

    def diagnostics(self, scope: str, name: str, indicator: str) -> dict[str, Any]:
        return _tools.diagnostics(scope=scope, name=name, indicator=indicator)

    def backtest(self, scope: str, name: str, indicator: str) -> dict[str, Any]:
        return _tools.backtest(scope=scope, name=name, indicator=indicator)

    def risk(self, scope: str, name: str, indicator: str) -> dict[str, Any]:
        return _tools.risk(scope=scope, name=name, indicator=indicator)

    def scenarios(self, scope: str, name: str, indicator: str, years: int = 5) -> dict[str, Any]:
        return _tools.scenarios(scope=scope, name=name, indicator=indicator, years=years)

    def monte_carlo(self, scope: str, name: str, indicator: str, years: int = 5) -> dict[str, Any]:
        return _tools.monte_carlo(scope=scope, name=name, indicator=indicator, years=years)

    def data_provenance(self, name: str) -> dict[str, Any]:
        return _tools.data_provenance(name=name)

    def macro_data(self) -> dict[str, Any]:
        return _tools.macro_data()

    def list_cities(self) -> dict[str, Any]:
        return _tools.list_cities()

    # LLM agent
    def chat(self, session_id: str, user_message: str) -> dict[str, Any]:
        """单轮 LLM agent。"""
        return _chat.chat(session_id=session_id, user_message=user_message)

    def chat_stream(self, session_id: str, user_message: str) -> Iterator[dict[str, Any]]:
        """流式 LLM agent。"""
        return _chat.chat_stream(session_id=session_id, user_message=user_message)

# cli/formatters.py
"""3 output formatters for CLI:
- format_json: machine-readable JSON
- format_plain: key=value,一行一对
- format_rich: 用 rich.Table 渲染人类可读

设计:
- 每个 formatter 接受 data dict,返 str(rich 返 Console capture)
- 不做 ANSI escape 检测 — 那是终端的事
- rich 暂不在本 task 实现,Task 11 处理
"""
from __future__ import annotations

import io
import json
from typing import Any


def format_json(data: dict[str, Any]) -> str:
    """JSON formatter — 完整 dict,ensure_ascii=False 保中文。"""
    return json.dumps(data, ensure_ascii=False, indent=2, default=str)


def format_plain(data: dict[str, Any]) -> str:
    """Plain formatter — key=value 行, 嵌套 dict 用 .join。

    例:
        scope=city
        name=深圳
        ensemble.predictions=[40344.36, 42273.63, 44202.90]
    """
    lines = []
    for k, v in _flatten(data):
        lines.append(f"{k}={v}")
    return "\n".join(lines)


def _flatten(data: dict[str, Any], prefix: str = "") -> list[tuple[str, Any]]:
    """递归展开嵌套 dict 成 (key.path, value) 列表。"""
    out = []
    for k, v in data.items():
        key = f"{prefix}.{k}" if prefix else k
        if isinstance(v, dict):
            out.extend(_flatten(v, prefix=key))
        elif isinstance(v, list):
            out.append((key, "[" + ", ".join(str(x) for x in v) + "]"))
        else:
            out.append((key, v))
    return out
```

def format_rich(data: dict[str, Any], command: str) -> str:
    """Rich formatter — 用 rich.Table 渲染,返 str(用 Console capture)。

    不用 rich.Live 不用 rich.Panel — 简单 Table。
    命令特定:
    - forecast: ensemble.predictions + CI + diagnostics
    - list-cities: 城市/省份两行
    - 其他: 嵌套 key=value 表
    """
    from rich.console import Console
    from rich.table import Table

    buf = io.StringIO()
    console = Console(file=buf, force_terminal=False, width=120, legacy_windows=False)

    if command == "forecast" and "ensemble" in data:
        _render_forecast_table(console, data)
    elif command == "list-cities" and "cities" in data:
        _render_cities_table(console, data)
    elif command == "risk" and "var_95" in data:
        _render_risk_table(console, data)
    elif command == "data-provenance" and "city" in data:
        _render_provenance_table(console, data)
    elif command == "macro-data" and "current" in data:
        _render_macro_table(console, data)
    else:
        _render_generic_kv(console, data, title=command)

    return console.export_text(styles=False)


def _render_forecast_table(console, data: dict) -> None:
    from rich.table import Table
    t = Table(title=f"{data.get('name', '?')} {data.get('indicator', '?').upper()} Forecast ({data.get('years', '?')}y)", show_header=True)
    t.add_column("Year", style="cyan")
    t.add_column("Prediction", justify="right", style="green")
    t.add_column("95% CI", justify="right")
    ensemble = data.get("ensemble", {}) or {}
    preds = ensemble.get("predictions", [])
    ci_lower = ensemble.get("ci_lower", [])
    ci_upper = ensemble.get("ci_upper", [])
    for i, p in enumerate(preds):
        ci = f"[{ci_lower[i]:.2f}, {ci_upper[i]:.2f}]" if i < len(ci_lower) and i < len(ci_upper) else "-"
        t.add_row(str(2024 + i) if "year" not in str(i) else str(i), f"{p:,.2f}", ci)
    console.print(t)
    if data.get("diagnostics_verdict"):
        console.print(f"[yellow]Diagnostics: {data['diagnostics_verdict']}[/yellow]")
    if data.get("backtest_mape") is not None:

# cli/main.py
"""Typer CLI 入口 — 11 个子命令 + global --format / --model / --base-url / --no-color。

设计:
- rca 是主命令(Typer app)
- 9 tool 子命令: forecast / diagnostics / backtest / risk / scenarios / monte-carlo /
  data-provenance / macro-data / list-cities
- 2 chat 子命令: chat / chat-stream
- global --format {rich,json,plain} (默认 rich)
- global --model X 覆盖 LLM
- global --base-url X 覆盖 gateway
- 错误处理: RCAError → exit 1; LLM 错 → exit 3
"""
from __future__ import annotations

import sys
from typing import Optional

import typer

from cli import formatters
from cli._resolve import (
    build_forecast_args,
)
from rca import tools as _rca_tools
from rca.errors import (
    LLMUnavailableError,
    RCAError,
    ToolArgumentError,
    ToolNotFoundError,
)

app = typer.Typer(
    name="rca",
    help="Regional Economic Analysis agent — CLI for 9 tools + LLM chat.",
    no_args_is_help=True,
    add_completion=False,
)

# --------------------------------------------------------------------------- #
# global state — 用 typer.Context 透传
# --------------------------------------------------------------------------- #


class State:
    format: str = "rich"
    model: Optional[str] = None
    base_url: Optional[str] = None
    no_color: bool = False


@app.callback()
def _main_callback(
    ctx: typer.Context,
    format: str = typer.Option("rich", "--format", help="Output format: rich / json / plain"),
    model: Optional[str] = typer.Option(None, "--model", help="Override LLM model (e.g. deepseek-v3)"),
    base_url: Optional[str] = typer.Option(None, "--base-url", help="Override LLM gateway base URL"),
    no_color: bool = typer.Option(False, "--no-color", help="Disable rich colors"),
) -> None:
    state = State()
    state.format = format
    state.model = model
    state.base_url = base_url
    state.no_color = no_color
    ctx.obj = state


def _get_state(ctx: typer.Context) -> State:
    return ctx.obj or State()


# --------------------------------------------------------------------------- #
# SDK invoke + 错误处理
# --------------------------------------------------------------------------- #


def _invoke_sdk(tool_name: str, kwargs: dict) -> dict:
    """调 rca.tools 的 tool,返 {"data": ..., "navigate_to": ...}。

    错误处理: 业务错 → exit 1; LLM 错 → exit 3。
    """
    func = getattr(_rca_tools, _TOOL_FUNC_NAMES[tool_name])
    try:
        return func(**kwargs)
    except ToolNotFoundError as e:
        typer.echo(f"ERROR: {e}", err=True)
        raise typer.Exit(code=1)
    except ToolArgumentError as e:
        typer.echo(f"ERROR: {e}", err=True)
        raise typer.Exit(code=1)
    except LLMUnavailableError as e:
        typer.echo(f"ERROR: LLM unavailable: {e}", err=True)
        raise typer.Exit(code=3)


# tool_name → rca.tools 函数名映射
_TOOL_FUNC_NAMES = {
    "get_forecast": "forecast",
    "get_diagnostics": "diagnostics",
    "get_backtest": "backtest",
    "get_risk": "risk",
    "get_scenarios": "scenarios",
    "get_monte_carlo": "monte_carlo",
    "get_data_provenance": "data_provenance",
    "get_macro_data": "macro_data",
    "get_list_cities": "list_cities",
}


def _render(ctx: typer.Context, data: dict, command: str) -> None:
    """按 global --format 渲染。"""
    state = _get_state(ctx)
    if state.format == "json":
        typer.echo(formatters.format_json(data))
    elif state.format == "plain":
        typer.echo(formatters.format_plain(data))
    else:  # rich
        typer.echo(formatters.format_rich(data, command=command))


# --------------------------------------------------------------------------- #
# 9 tool 子命令 — 本 task 实现 forecast
# --------------------------------------------------------------------------- #


@app.command()
def forecast(
    ctx: typer.Context,
    city: str = typer.Option(..., "--city", help="城市拼音 (shenzhen) / 省份名 (guangdong)"),
    indicator: str = typer.Option(..., "--indicator", help="指标: gdp / population / ..."),
    scope: str = typer.Option("city", "--scope", help="city 或 province"),
    years: Optional[int] = typer.Option(None, "--years", help="预测年数 (默认 5)"),
) -> None:
    """多年预测 ensemble + diagnostics + backtest 摘要。"""
    args = build_forecast_args(city=city, indicator=indicator, scope=scope, years=years)
    result = _invoke_sdk("get_forecast", args)
    _render(ctx, result["data"], command="forecast")

# Append to cli/main.py
from rca import chat as _rca_chat_module
from rca import chat_stream as _rca_chat_stream_module  # not exposed — use module


def _rca_chat(session_id: str, user_message: str) -> dict:
    return _rca_chat_module.chat(session_id=session_id, user_message=user_message)


def _rca_chat_stream(session_id: str, user_message: str):
    return _rca_chat_module.chat_stream(session_id=session_id, user_message=user_message)


@app.command()
def chat(
    ctx: typer.Context,
    message: str = typer.Option(..., "--message", "-m", help="用户问题"),
    session: str = typer.Option("cli-default", "--session", "-s", help="会话 ID(保留多轮上下文)"),
) -> None:
    """单轮 LLM agent — 自然语言 → 调 tool → 返答案。"""
    state = _get_state(ctx)
    try:
        result = _rca_chat(session_id=session, user_message=message)
    except LLMUnavailableError as e:
        typer.echo(f"ERROR: LLM unavailable: {e}", err=True)
        raise typer.Exit(code=3)
    except RCAError as e:
        typer.echo(f"ERROR: {e}", err=True)
        raise typer.Exit(code=1)
    if state.format == "json":
        typer.echo(formatters.format_json(result))
    else:
        # rich / plain 模式: 渲染 content + 简短 tool_rounds 信息
        typer.echo(result.get("content", "(no content)"))
        if result.get("tool_rounds"):
            typer.echo(f"\n[tool_rounds: {result['tool_rounds']}]", err=True)


@app.command(name="chat-stream")
def chat_stream(
    ctx: typer.Context,
    message: str = typer.Option(..., "--message", "-m", help="用户问题"),
    session: str = typer.Option("cli-default", "--session", "-s"),
) -> None:
    """流式 LLM agent — 实时显示 thinking → tool_call → token 增量。"""
    try:
        for ev in _rca_chat_stream(session_id=session, user_message=message):
            et = ev.get("type")
            if et == "thinking":
                typer.echo("[thinking]...", err=True)
            elif et == "tool_call":
                typer.echo(f"[tool_call] {ev.get('name')}({ev.get('args')})", err=True)
            elif et == "token":
                typer.echo(ev.get("content", ""), nl=False)
            elif et == "done":
                typer.echo(f"\n[done: rounds={ev.get('tool_rounds', 0)}]", err=True)
            elif et == "error":
                typer.echo(f"\n[error: {ev.get('message', '')[:200]}]", err=True)
                raise typer.Exit(code=3)
    except LLMUnavailableError as e:
        typer.echo(f"\nERROR: LLM unavailable: {e}", err=True)
        raise typer.Exit(code=3)
    except RCAError as e:
        typer.echo(f"\nERROR: {e}", err=True)
        raise typer.Exit(code=1)
    typer.echo("")  # 末尾换行
